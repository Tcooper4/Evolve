84	def _load_universe(name: str, fallback: list) -> list:
85	    _dir = os.path.dirname(os.path.abspath(__file__))
86	    json_path = os.path.join(_dir, "..", "data", "universes", f"{name}.json")
87	    try:
88	        with open(json_path, encoding="utf-8", errors="replace") as f:
89	            tickers = json.load(f)
90	        if tickers:
91	            return tickers
92	    except Exception as _e:
93	        logger.warning("Scanner: universe load failed (json) for %s: %s", name, _e)
94	        st.caption(f"âš ï¸ Universe load failed: {_e}")
95	    try:
96	        if name == "sp500":
97	            url = "https://en.wikipedia.org/wiki/List_of_S%26P_500_companies"
98	            tables = pd.read_html(url)
99	            if tables:
100	                table = tables[0]
101	                tickers = (
102	                    table[[c for c in table.columns if str(c).lower() in ("symbol", "ticker")][0]]
103	                    .astype(str)
104	                    .str.replace(".", "-", regex=False)
105	                    .str.upper()
106	                    .tolist()
107	                )
108	                if tickers:
109	                    return tickers
110	        elif name in ("russell1000", "russell3000"):
111	            url = "https://en.wikipedia.org/wiki/Russell_3000_Index"
112	            tables = pd.read_html(url)
113	            tickers_list = []
114	            for t in tables:
115	                cols = [c.lower() for c in t.columns.astype(str)]
116	                if any("ticker" in c or "symbol" in c for c in cols):
117	                    for col in t.columns:
118	                        if "ticker" in str(col).lower() or "symbol" in str(col).lower():
119	                            series = t[col].astype(str).str.replace(".", "-", regex=False).str.upper()
120	                            tickers_list.extend(series.tolist())
121	                    break
122	            tickers_list = [t for t in tickers_list if t and t != "nan"]
123	            if tickers_list:
124	                return tickers_list[:1000] if name == "russell1000" else tickers_list
125	        elif name == "sp500_nasdaq100":
126	            sp500 = _load_universe("sp500", fallback)
127	            nasdaq100 = _load_universe("nasdaq100", fallback)
128	            if sp500 or nasdaq100:
129	                return sorted(set(sp500 or []).union(nasdaq100 or []))
130	    except Exception as _e:
131	        logger.warning(
132	            "Scanner: universe load failed (web) for %s: %s",
133	            name,
134	            _e,
135	        )
136	        st.caption(f"âš ï¸ Universe load failed: {_e}")
137	    return fallback

140	def _get_short_float(ticker: str) -> str:
141	    try:
142	        import yfinance as yf
143	
144	        info = yf.Ticker(ticker).info
145	        pct = info.get("shortPercentOfFloat", None)
146	        if pct is None:
147	            return "N/A"
148	        return f"{pct * 100:.1f}%"
149	    except Exception:
150	        return "N/A"

171	def _news_score(ticker: str) -> tuple:
172	    """Return (label, color) for News Score: HOT/POS/NEU/NEG."""
173	    pos_kw = ["surge", "beat", "rise", "gain", "growth", "up", "bull", "upgrade", "strong"]
174	    neg_kw = ["fall", "drop", "miss", "cut", "down", "bear", "downgrade", "weak", "loss"]
175	    try:
176	        items = get_news(ticker)
177	        now = datetime.utcnow()
178	        score = 0
179	        for item in items:
180	            title = (item.get("title") or item.get("headline") or "").lower()
181	            pub = item.get("published") or item.get("providerPublishTime")
182	            if pub:
183	                try:
184	                    if isinstance(pub, (int, float)):
185	                        dt = datetime.utcfromtimestamp(pub / 1000 if pub > 1e12 else pub)
186	                    else:
187	                        dt = datetime.fromisoformat(str(pub).replace("Z", "+00:00"))
188	                    if (now - dt.replace(tzinfo=None) if getattr(dt, "tzinfo", None) else now - dt).days > 1:
189	                        continue
190	                except Exception:
191	                    pass
192	            score += sum(1 for w in pos_kw if w in title) - sum(1 for w in neg_kw if w in title)
193	        if score >= 3:
194	            return "HOT", "#ff9800"
195	        if score >= 1:
196	            return "POS", "#26a69a"
197	        if score <= -1:
198	            return "NEG", "#ef5350"
199	        return "NEU", "#4a6080"
200	    except Exception:
201	        return "NEU", "#4a6080"
