32	@st.cache_data(ttl=900)
33	def get_market_pulse() -> dict:
34	    tickers = {
35	        "SPY": "S&P 500",
36	        "^VIX": "VIX",
37	    }
38	    out = {}
39	    for tkr, label in tickers.items():
40	        try:
41	            q = get_quote(tkr)
42	            price = q.get("price")
43	            if price is None:
44	                h = get_history(tkr, period="1d")
45	                if not h.empty:
46	                    price = float(h["Close"].iloc[-1])
47	            prev = q.get("prev_close")
48	            if prev in (None, 0) and tkr != "^VIX":
49	                h2 = get_history(tkr, period="5d")
50	                if len(h2) >= 2:
51	                    prev = float(h2["Close"].iloc[-2])
52	            if price is None or prev in (None, 0):
53	                continue
54	            chg = (float(price) - float(prev)) / float(prev) * 100
55	            out[tkr] = {"label": label, "price": float(price), "chg": chg}
56	        except Exception:
57	            continue
58	    return out

354	def _build_fallback_digest(scores: list) -> str:
355	    """Plain text digest when LLM is unavailable."""
356	    if not scores:
357	        return "No watchlist data."
358	    _top = scores[0]
359	    _bot = scores[-1]
360	    _bullish = [s for s in scores if s["score"] >= 6.5]
361	    _bearish = [s for s in scores if s["score"] <= 4.0]
362	    _lines = []
363	    if _bullish:
364	        _syms = ", ".join(s["symbol"] for s in _bullish[:3])
365	        _lines.append(f"**Strongest setups:** {_syms}")
366	    if _bearish:
367	        _syms = ", ".join(s["symbol"] for s in _bearish[:3])
368	        _lines.append(f"**Weak signals:** {_syms}")
369	    _lines.append(
370	        f"**Top pick:** {_top['symbol']} "
371	        f"(score {_top['score']:.1f}, grade {_top['grade']})"
372	    )
373	    return " Â· ".join(_lines) if _lines else "No clear setups."
