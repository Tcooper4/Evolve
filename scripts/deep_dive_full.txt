20	def _deep_dive_platform_suffix(
21	    sym: str,
22	    price,
23	    chg,
24	    score,
25	    rec,
26	) -> str:
27	    """Pre-loaded ticker context for tool router + synthesis (no tool execution here)."""
28	    parts = [f"Deep dive symbol: {sym}"]
29	    if price is not None:
30	        chg_s = f"{float(chg):+.2f}%" if chg is not None else "â€”"
31	        parts.append(f"Last price: ${float(price):.2f} Â· session change {chg_s}")
32	    if score and not score.get("error"):
33	        parts.append(
34	            f"AI score: overall {score.get('overall_score')} grade {score.get('grade')} "
35	            f"(T/M/S/F: {score.get('technical_score')}/"
36	            f"{score.get('momentum_score')}/{score.get('sentiment_score')}/"
37	            f"{score.get('fundamental_score')})"
38	        )
39	        summ = score.get("summary") or ""
40	        if summ:
41	            parts.append(f"Score summary: {summ[:600]}")
42	    sigs = (score or {}).get("signals") or []
43	    if sigs:
44	        parts.append("Top signals:")
45	        for s in sigs[:5]:
46	            parts.append(
47	                f"  - [{s.get('impact')}] {s.get('name')}: {s.get('value')}"
48	            )
49	    if rec:
50	        parts.append(
51	            f"Recommendation snapshot: {rec.get('action')} Â· "
52	            f"conviction {rec.get('conviction')} Â· entry {rec.get('entry')} Â· "
53	            f"target {rec.get('target')} Â· stop {rec.get('stop')}"
54	        )
55	    try:
56	        from trading.data.earnings_calendar import get_upcoming_earnings
57	
58	        er = get_upcoming_earnings(sym)
59	        if er and er.get("days_until") is not None:
60	            parts.append(f"Upcoming earnings: in {er.get('days_until')} days")
61	    except Exception:
62	        pass
63	    try:
64	        from trading.data.news_aggregator import get_news
65	
66	        items = get_news(sym, max_items=5)
67	        if items:
68	            parts.append("Recent headlines (snapshot):")
69	            for it in items[:5]:
70	                t = (it.get("title") or it.get("headline") or str(it))[:200]
71	                parts.append(f"  - {t}")
72	    except Exception:
73	        pass
74	    return "\n".join(parts)

77	def _render_deep_dive_chat(
78	    sym: str,
79	    price,
80	    chg,
81	    score,
82	    rec,
83	) -> None:
84	    """Bottom chat bar: same tool loop as Home, with ticker-heavy context."""
85	    st.markdown("---")
86	    st.subheader(f"Ask about {sym}")
87	    _dk = f"deep_dive_chat_{sym}"
88	    if _dk not in st.session_state:
89	        st.session_state[_dk] = []
90	
91	    platform_suffix = _deep_dive_platform_suffix(sym, price, chg, score, rec)
92	    _inp = st.chat_input(f"Ask about {sym}...", key=f"dd_chat_in_{sym}")
93	    _raw = (_inp or "").strip()
94	
95	    if _raw:
96	        st.session_state[_dk].append({"role": "user", "content": _raw})
97	
98	    for msg in st.session_state[_dk]:
99	        with st.chat_message(msg.get("role", "user")):
100	            if msg.get("role") == "assistant":
101	                for _c in msg.get("tool_captions") or []:
102	                    st.caption(_c)
103	            st.markdown(msg.get("content", ""))
104	
105	    if not _raw:
106	        return
107	
108	    res = None
109	    with st.chat_message("assistant"):
110	        with st.spinner("Thinking..."):
111	            try:
112	                from agents.llm.tool_executor import execute_with_tools
113	                from trading.memory import get_memory_store
114	                from trading.services import chat_nl_service
115	
116	                store = get_memory_store()
117	                um = f"[Deep dive: {sym}]\n{_raw}"
118	                try:
119	                    store.ingest_preference_text(um, source="deep_dive_chat")
120	                except Exception as e:
121	                    logger.warning("Deep dive chat: preference ingest failed: %s", e)
122	                if "chat_router" not in st.session_state:
123	                    try:
124	                        from trading.agents.enhanced_prompt_router import (
125	                            EnhancedPromptRouterAgent,
126	                        )
127	
128	                        st.session_state.chat_router = EnhancedPromptRouterAgent()
129	                    except Exception as e:
130	                        logger.warning("Deep dive chat: router failed: %s", e)
131	                        st.session_state.chat_router = None
132	                router = st.session_state.get("chat_router")
133	                route_result = (
134	                    chat_nl_service.parse_intent(router, um)
135	                    if router
136	                    else {"intent": "unknown", "args": {}}
137	                )
138	                intent = (
139	                    route_result.get("intent", "unknown")
140	                    if isinstance(route_result, dict)
141	                    else getattr(route_result, "intent", "unknown")
142	                )
143	                memory_context = chat_nl_service.get_memory_context(store)
144	                agent_response = chat_nl_service.run_agent_action(um)
145	                context_block = chat_nl_service.build_context_block(
146	                    memory_context,
147	                    agent_response,
148	                    intent=intent,
149	                    store=store,
150	                )
151	                conv = [
152	                    {"role": m["role"], "content": m.get("content", "")}
153	                    for m in st.session_state[_dk][:-1]
154	                ]
155	                res = execute_with_tools(
156	                    user_message=um,
157	                    context_block=context_block,
158	                    conversation_messages=conv,
159	                    system_prompt=chat_nl_service.EVOLVE_CHAT_SYSTEM_PROMPT,
160	                    platform_context_suffix=platform_suffix,
161	                    focus_symbol=sym,
162	                    available_tools=[
163	                        "scan_universe",
164	                        "get_ai_score",
165	                        "get_forecast",
166	                        "get_news",
167	                        "get_risk_metrics",
168	                    ],
169	                    max_tokens=2048,
170	                )
171	                for _c in res.tool_captions or []:
172	                    st.caption(_c)
173	                st.markdown(res.text)
174	            except Exception as e:
175	                logger.exception("Deep dive chat failed: %s", e)
176	                from agents.llm.tool_executor import ToolChatResult
177	
178	                res = ToolChatResult(
179	                    text=f"Chat unavailable: {e}",
180	                    tool_captions=[],
181	                )
182	                st.caption(res.text)
183	    st.session_state[_dk].append(
184	        {
185	            "role": "assistant",
186	            "content": res.text if res else "",
187	            "tool_captions": (res.tool_captions or []) if res else [],
188	        }
189	    )
