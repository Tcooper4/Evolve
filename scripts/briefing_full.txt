26	class MorningBriefing:
27	    """
28	    Autonomous morning briefing pipeline.
29	
30	    Steps:
31	    1. Scan universe (SP100 by default) for top AI scores
32	    2. Run consensus forecast on top candidates
33	    3. Check short interest and earnings calendar
34	    4. Generate structured report with entry/target/stop
35	    5. Summarize market regime
36	    """

71	    def _opportunity_passes_direction_pref(self, opp: Dict[str, Any]) -> bool:
72	        prefs = getattr(self, "_briefing_prefs", {}) or {}
73	        od = prefs.get(
74	            "opportunity_direction",
75	            "Bullish only (BUY signals)",
76	        )
77	        fc = opp.get("forecast") or {}
78	        entry = float(opp.get("entry") or 0)
79	        target = float(opp.get("target") or fc.get("consensus_price") or 0)
80	        MIN_MOVE_PCT = 0.005
81	        if target > 0 and entry > 0:
82	            move_pct = abs(target - entry) / entry
83	            if move_pct < MIN_MOVE_PCT:
84	                return False
85	        exp_pct = float(fc.get("expected_move_pct", 0) or 0) / 100.0
86	        if entry > 0 and target > 0:
87	            forecast_return = (target - entry) / entry
88	        else:
89	            forecast_return = exp_pct
90	
91	        sym = str(opp.get("symbol") or "?")
92	
93	        if "Both directions" in od:
94	            return True
95	        if "Bearish only" in od:
96	            is_bear = forecast_return < -0.005 or (
97	                target > 0 and entry > 0 and target < entry * 0.998
98	            )
99	            if not is_bear:
100	                logger.info(
101	                    "Skipping %s: not bearish enough for bearish-only pref "
102	                    "(return=%.2f%%)",
103	                    sym,
104	                    forecast_return * 100.0,
105	                )
106	            return is_bear
107	
108	        is_bull = forecast_return > 0.005 or (
109	            target > 0 and entry > 0 and target > entry * 1.002
110	        )
111	        if not is_bull:
112	            logger.info(
113	                "Skipping %s: bearish forecast (return=%.2f%%)",
114	                sym,
115	                forecast_return * 100.0,
116	            )
117	        return is_bull

119	    def generate(
120	        self,
121	        progress_callback: Optional[Callable[[int, int], None]] = None,
122	    ) -> Dict[str, Any]:
123	        """
124	        Generate full morning briefing.
125	        Returns dict with markdown report and structured data.
126	
127	        progress_callback: optional callable(done, total) during universe scan.
128	        """
129	        logger.info("Morning briefing: starting generation")
130	        report = {
131	            "timestamp": datetime.now().isoformat(),
132	            "market_regime": {},
133	            "top_opportunities": [],
134	            "watchlist_alerts": [],
135	            "risk_summary": {},
136	            "markdown": "",
137	            "error": None,
138	        }
139	
140	        try:
141	            # Step 1: Market regime
142	            report["market_regime"] = self._get_market_regime()
143	
144	            # Step 2: Scan universe
145	            candidates = self._scan_universe(
146	                progress_callback=progress_callback
147	            )
148	            self._briefing_prefs = self._load_briefing_prefs()
149	            pref_sectors = self._briefing_prefs.get("preferred_sectors") or []
150	            if pref_sectors and candidates:
151	                try:
152	                    from trading.data.price_cache import get_info
153	
154	                    filtered_c: List[Dict[str, Any]] = []
155	                    for c in candidates:
156	                        sym = c.get("symbol")
157	                        if not sym:
158	                            continue
159	                        try:
160	                            info = get_info(str(sym)) or {}
161	                            sec_raw = (info.get("sector") or "").strip()
162	                        except Exception:
163	                            sec_raw = ""
164	                        sec_l = sec_raw.lower()
165	                        if not sec_l:
166	                            continue
167	                        if any(
168	                            str(s).strip().lower() in sec_l
169	                            for s in pref_sectors
170	                        ):
171	                            nc = dict(c)
172	                            nc["sector"] = sec_raw
173	                            filtered_c.append(nc)
174	                    if filtered_c:
175	                        candidates = filtered_c
176	                        logger.info(
177	                            "Sector filter applied: %d candidates in %s",
178	                            len(candidates),
179	                            pref_sectors,
180	                        )
181	                except Exception as _sf:
182	                    logger.debug("Sector filter skipped: %s", _sf)
183	
184	            logger.info(
185	                "Morning briefing: found %d candidates above %.1f score",
186	                len(candidates),
187	                self.min_ai_score,
188	            )
189	
190	            # Step 3: Deep analysis â€” try extra names until slots filled
191	            opportunities = []
192	            shared_router = None
193	            try:
194	                from trading.models.forecast_router import (
195	                    ForecastRouter,
196	                    get_router_singleton,
197	                )
198	
199	                shared_router = get_router_singleton()
200	            except Exception as e:
201	                logger.warning("Morning briefing: ForecastRouter init failed: %s", e)
202	
203	            _max_try = min(len(candidates), max(self.max_positions * 6, 24))
204	            _pool = candidates[:_max_try]
205	            _n_pool = len(_pool)
206	            for i, candidate in enumerate(_pool):
207	                if len(opportunities) >= self.max_positions:
208	                    break
209	                sym = candidate.get("symbol") or "?"
210	                t0 = time.perf_counter()
211	                logger.info(
212	                    "Morning briefing: analyzing %s (%d/%d)",
213	                    sym,
214	                    i + 1,
215	                    _n_pool,
216	                )
217	                try:
218	                    opp = self._analyze_opportunity(
219	                        candidate,
220	                        router=shared_router,
221	                        briefing=True,
222	                    )
223	                    if opp and self._opportunity_passes_direction_pref(opp):
224	                        opportunities.append(opp)
225	                except Exception as e:
226	                    logger.debug(
227	                        "Opportunity analysis failed for %s: %s",
228	                        sym,
229	                        e,
230	                    )
231	                elapsed = time.perf_counter() - t0
232	                logger.info(
233	                    "Morning briefing: %s complete (%.1fs)",
234	                    sym,
235	                    elapsed,
236	                )
237	
238	            if len(opportunities) >= 2:
239	                try:
240	                    from trading.data.price_cache import get_history
241	                    from trading.optimization.portfolio_optimizer import (
242	                        get_portfolio_optimizer,
243	                    )
244	
245	                    _opt = get_portfolio_optimizer()
246	                    _syms = [
247	                        str(o.get("symbol", "")).upper()
248	                        for o in opportunities
249	                    ]
250	                    _returns_dict: Dict[str, pd.Series] = {}
251	                    for sym in _syms:
252	                        try:
253	                            h = get_history(
254	                                sym, period="126d", interval="1d"
255	                            )
256	                            if h is not None and len(h) >= 30:
257	                                col = {
258	                                    c.lower(): c for c in h.columns
259	                                }.get("close", h.columns[0])
260	                                _returns_dict[sym] = (
261	                                    h[col].pct_change().dropna()
262	                                )
263	                        except Exception:
264	                            continue
265	                    if len(_returns_dict) >= 2:
266	                        _returns_df = pd.DataFrame(_returns_dict).dropna(
267	                            how="any"
268	                        )
269	                        if len(_returns_df) >= 30:
270	                            _result = _opt.mean_variance_optimization(
271	                                _returns_df, target_return=None
272	                            )
273	                            if (
274	                                isinstance(_result, dict)
275	                                and "weights" in _result
276	                                and "error" not in _result
277	                            ):
278	                                _weights = _result["weights"]
279	                                for opp in opportunities:
280	                                    sym = opp["symbol"]
281	                                    w = float(
282	                                        _weights.get(
283	                                            sym, 1.0 / len(_syms)
284	                                        )
285	                                    )
286	                                    opp["suggested_weight"] = round(w, 3)
287	                                    opp["weight_pct"] = f"{w * 100:.0f}%"
288	                                report["portfolio"] = {
289	                                    "method": "max_sharpe",
290	                                    "expected_sharpe": _result.get(
291	                                        "sharpe_ratio"
292	                                    ),
293	                                    "weights": _weights,
294	                                    "note": (
295	                                        "Mean-variance optimized weights "
296	                                        "(6-month history)"
297	                                    ),
298	                                }
299	                                logger.info(
300	                                    "Portfolio optimized: %s", _weights
301	                                )
302	                except Exception as e:
303	                    logger.debug("Portfolio opt failed: %s", e)
304	
305	            report["top_opportunities"] = opportunities
306	
307	            # Step 4: Watchlist alerts
308	            report["watchlist_alerts"] = self._get_watchlist_alerts()
309	
310	            # Step 5: Generate markdown
311	            report["markdown"] = self._format_markdown(report)
312	
313	            self._last_report = report
314	            logger.info("Morning briefing: complete")
315	
316	        except Exception as e:
317	            logger.warning("Morning briefing generation failed: %s", e)
318	            report["error"] = str(e)
319	            report["markdown"] = (
320	                f"âš ï¸ Briefing generation encountered an error: {e}\n\n"
321	                "Please check the Scanner page for manual analysis."
322	            )
323	
324	        return report

326	    def _get_market_regime(self) -> Dict[str, Any]:
327	        """Get current market regime from SPY/QQQ/VIX."""
328	        regime = {
329	            "spy_trend": "UNKNOWN",
330	            "vix_level": None,
331	            "regime": "NEUTRAL",
332	            "description": "",
333	        }
334	        try:
335	            import numpy as np
336	            from trading.data.price_cache import get_history, get_macro_history
337	
338	            spy = get_history("SPY", period="3mo")
339	            vix_hist = get_macro_history("^VIX", period="5d")
340	
341	            if not spy.empty:
342	                _col_map = {c.lower(): c for c in spy.columns}
343	                close_col = _col_map.get("close", spy.columns[0])
344	                closes = spy[close_col].values
345	                sma50 = np.mean(closes[-50:]) if len(closes) >= 50 else closes.mean()
346	                current = float(closes[-1])
347	                regime["spy_trend"] = "BULLISH" if current > sma50 else "BEARISH"
348	                regime["spy_50d_pct"] = round(
349	                    (current - sma50) / sma50 * 100, 2
350	                )
351	
352	            if not vix_hist.empty:
353	                _col_map = {c.lower(): c for c in vix_hist.columns}
354	                close_col = _col_map.get("close", vix_hist.columns[0])
355	                vix_level = float(vix_hist[close_col].iloc[-1])
356	                regime["vix_level"] = round(vix_level, 2)
357	
358	                if vix_level > 30:
359	                    regime["volatility"] = "HIGH"
360	                elif vix_level > 20:
361	                    regime["volatility"] = "ELEVATED"
362	                else:
363	                    regime["volatility"] = "NORMAL"
364	
365	            # Determine overall regime (None = unknown VIX â†’ neutral 20 for thresholds; keep real 0.0)
366	            is_bullish = regime["spy_trend"] == "BULLISH"
367	            vix_val = regime["vix_level"]
368	            if vix_val is None:
369	                vix_n = 20.0
370	            else:
371	                vix_n = float(vix_val)
372	            if is_bullish and vix_n < 20:
373	                regime["regime"] = "RISK_ON"
374	                regime["description"] = "Market is in risk-on mode â€” momentum strategies favored"
375	            elif not is_bullish and vix_n > 25:
376	                regime["regime"] = "RISK_OFF"
377	                regime["description"] = "Market is in risk-off mode â€” defensive positioning recommended"
378	            else:
379	                regime["regime"] = "NEUTRAL"
380	                regime["description"] = "Mixed signals â€” selective positioning recommended"
381	
382	        except Exception as e:
383	            logger.debug("Market regime fetch failed: %s", e)
384	            regime["error"] = str(e)
385	
386	        return regime

388	    def _scan_universe(
389	        self,
390	        progress_callback: Optional[Callable[[int, int], None]] = None,
391	    ) -> List[Dict[str, Any]]:
392	        """Run scanner on universe and return top candidates (symbol + ai_score)."""
393	        try:
394	            from trading.analysis.market_scanner import _get_universe, scan_market
395	
396	            u = (self.universe or "default").lower()
397	            uni = list(_get_universe(u))
398	            try:
399	                from config.user_store import load_user_preferences
400	                from utils.session_utils import get_stable_user_id
401	
402	                _uid = get_stable_user_id()
403	                _p = load_user_preferences(_uid) or {}
404	                if _p.get("min_ai_score") is not None:
405	                    self.min_ai_score = float(_p["min_ai_score"])
406	                pref_uni = str(_p.get("briefing_universe") or "")
407	                if "NASDAQ100" in pref_uni:
408	                    uni = list(_get_universe("nasdaq100"))
409	                elif "SP500" in pref_uni:
410	                    uni = list(_get_universe("sp500"))[:50]
411	                elif "SP100" in pref_uni:
412	                    uni = list(_get_universe("sp100"))
413	                elif "Top 25" in pref_uni:
414	                    uni = list(_get_universe("sp100"))[:25]
415	                if _p.get("watchlist_only"):
416	                    try:
417	                        from trading.data.watchlist import WatchlistManager
418	
419	                        _wl = WatchlistManager().get_all() or []
420	                        _syms = [
421	                            str(r.get("symbol", "")).strip().upper()
422	                            for r in _wl
423	                            if r.get("symbol")
424	                        ]
425	                        if _syms:
426	                            uni = _syms
427	                            logger.info(
428	                                "Morning briefing: using watchlist (%d tickers)",
429	                                len(uni),
430	                            )
431	                    except Exception as _wl_e:
432	                        logger.debug("Watchlist-only universe skipped: %s", _wl_e)
433	            except Exception:
434	                pass
435	            if u in ("sp50", "large", "mega"):
436	                uni = uni[:50]
437	            elif u in ("sp30", "core"):
438	                uni = uni[:30]
439	            uni = uni[: self.BRIEFING_UNIVERSE_CAP]
440	
441	            logger.info(
442	                "Morning briefing: scanning %d tickers...",
443	                len(uni),
444	            )
445	            # tz_localize(None) is naive-only; batch tz strip is in market_scanner.scan_market
446	
447	            raw = scan_market(
448	                filters=[],
449	                universe=uni,
450	                max_results=min(
451	                    200,
452	                    max(len(uni), self.max_positions * 16),
453	                ),
454	                progress_callback=progress_callback,
455	            )
456	            if raw.get("error"):
457	                logger.warning("Universe scan error: %s", raw.get("error"))
458	                return []
459	            rows = raw.get("results") or []
460	            candidates = [
461	                r
462	                for r in rows
463	                if float(r.get("ai_score", 0) or 0) >= float(self.min_ai_score)
464	            ]
465	            candidates.sort(
466	                key=lambda x: float(x.get("ai_score", 0) or 0),
467	                reverse=True,
468	            )
469	            return candidates[: self.max_positions * 2]
470	
471	        except Exception as e:
472	            logger.warning("Universe scan failed: %s", e)
473	            return []

475	    def _analyze_opportunity(
476	        self,
477	        candidate: Dict[str, Any],
478	        router: Any = None,
479	        briefing: bool = True,
480	    ) -> Optional[Dict[str, Any]]:
481	        """Deep analysis on a single candidate.
482	
483	        When ``briefing`` is True, skip Monte Carlo and strategy comparison
484	        (heavy paths reserved for Analyze deep dive).
485	        """
486	        symbol = candidate.get("symbol")
487	        if not symbol:
488	            return None
489	
490	        opp = {
491	            "symbol": symbol,
492	            "ai_score": candidate.get("ai_score"),
493	            "current_price": None,
494	            "forecast": {},
495	            "entry": None,
496	            "target": None,
497	            "stop": None,
498	            "conviction": "MEDIUM",
499	            "thesis": "",
500	            "risks": [],
501	            "catalysts": [],
502	        }
503	
504	        try:
505	            from trading.data.price_cache import get_history
506	
507	            hist = get_history(str(symbol), period="6mo")
508	
509	            if hist.empty:
510	                return None
511	
512	            _col_map = {c.lower(): c for c in hist.columns}
513	            close_col = _col_map.get("close", hist.columns[0])
514	            current_price = float(hist[close_col].iloc[-1])
515	            opp["current_price"] = round(current_price, 2)
516	
517	            # Get consensus forecast
518	            try:
519	                from trading.models.forecast_router import (
520	                    ForecastRouter,
521	                    get_router_singleton,
522	                )
523	
524	                _router = router if router is not None else get_router_singleton()
525	                forecast = _router.get_consensus_forecast(
526	                    data=hist,
527	                    horizon=7,
528	                    symbol=str(symbol),
529	                    models=self.BRIEFING_MODELS,
530	                    model_configs={"arima": {"fast_mode": True}},
531	                )
532	                if forecast and "error" not in forecast:
533	                    consensus_price = forecast.get("consensus_price")
534	                    if consensus_price:
535	                        pct_move = (
536	                            (consensus_price - current_price)
537	                            / current_price * 100
538	                        )
539	                        opp["forecast"] = {
540	                            "consensus_price": round(consensus_price, 2),
541	                            "expected_move_pct": round(pct_move, 2),
542	                            "direction": forecast.get("direction", "NEUTRAL"),
543	                            "conviction": forecast.get("conviction", "LOW"),
544	                            "models_used": forecast.get("models_used", []),
545	                        }
546	
547	                        # Entry/target/stop
548	                        opp["entry"] = round(current_price, 2)
549	                        opp["target"] = round(consensus_price, 2)
550	                        stop_pct = 0.03  # 3% stop
551	                        if forecast.get("direction") == "BULLISH":
552	                            opp["stop"] = round(
553	                                current_price * (1 - stop_pct), 2
554	                            )
555	                        else:
556	                            opp["stop"] = round(
557	                                current_price * (1 + stop_pct), 2
558	                            )
559	
560	                        opp["conviction"] = forecast.get("conviction", "MEDIUM")
561	                    for _w in forecast.get("walk_forward_warnings") or []:
562	                        opp.setdefault("risks", []).append(_w)
563	                    _wfc = forecast.get("walk_forward_confidence")
564	                    if _wfc:
565	                        opp.setdefault("catalysts", []).append(
566	                            f"Walk-forward confidence: {_wfc}"
567	                        )
568	            except Exception as e:
569	                logger.debug("Forecast failed for %s: %s", symbol, e)
570	
571	            if briefing:
572	                try:
573	                    from trading.backtesting.monte_carlo import (
574	                        MonteCarloConfig,
575	                        MonteCarloSimulator,
576	                    )
577	
578	                    import numpy as np
579	
580	                    _cm = {c.lower(): c for c in hist.columns}
581	                    _cc = _cm.get("close", hist.columns[0])
582	                    _rets = hist[_cc].astype(float).pct_change().dropna()
583	                    if len(_rets) >= 30:
584	                        _cfg = MonteCarloConfig(n_simulations=50)
585	                        _mc = MonteCarloSimulator(_cfg)
586	                        _paths = _mc.simulate_portfolio_paths(
587	                            _rets, n_simulations=50
588	                        )
589	                        if _paths is not None and len(_paths) > 1:
590	                            _first = _paths.iloc[0].values.astype(float)
591	                            _last = _paths.iloc[-1].values.astype(float)
592	                            _term = (_last / np.maximum(_first, 1e-12)) - 1.0
593	                            _p10 = float(np.percentile(_term, 10))
594	                            if _p10 < -0.05:
595	                                opp["risk_note"] = (
596	                                    f"âš ï¸ Downside risk: 10th pct = {_p10:.1%}"
597	                                )
598	                except Exception:
599	                    pass
600	
601	            if not briefing:
602	                try:
603	                    from trading.backtesting.monte_carlo import (
604	                        MonteCarloConfig,
605	                        MonteCarloSimulator,
606	                    )
607	
608	                    import numpy as np
609	
610	                    _cfg = MonteCarloConfig(n_simulations=100)
611	                    _mc = MonteCarloSimulator(_cfg)
612	                    _cm = {c.lower(): c for c in hist.columns}
613	                    _cc = _cm.get("close", hist.columns[0])
614	                    _rets = hist[_cc].astype(float).pct_change().dropna()
615	                    if len(_rets) >= 30:
616	                        _paths = _mc.simulate_portfolio_paths(
617	                            _rets, n_simulations=100
618	                        )
619	                        if _paths is not None and len(_paths) > 1:
620	                            _first = _paths.iloc[0].values.astype(float)
621	                            _last = _paths.iloc[-1].values.astype(float)
622	                            _term = (_last / np.maximum(_first, 1e-12)) - 1.0
623	                            _p10 = float(np.percentile(_term, 10))
624	                            _fmove = float(
625	                                (opp.get("forecast") or {}).get(
626	                                    "expected_move_pct", 0
627	                                )
628	                                or 0
629	                            )
630	                            if _p10 < -0.08:
631	                                opp["risk_note"] = (
632	                                    f"High downside risk: 10th pct = "
633	                                    f"{_p10:.1%} (vs ~{_fmove:+.1f}% "
634	                                    f"consensus move)"
635	                                )
636	                except Exception as _mce:

709	        """Check watchlist for notable moves or signals."""
710	        alerts = []
711	        try:
712	            from trading.data.watchlist import WatchlistManager
713	
714	            _rows = WatchlistManager().get_all() or []
715	            watchlist = [
716	                str(r.get("symbol", "")).strip().upper()
717	                for r in _rows
718	                if r.get("symbol")
719	            ]
720	            if not watchlist:
721	                return []
722	
723	            import numpy as np
724	            from trading.data.price_cache import get_history
725	
726	            for symbol in watchlist[:10]:
727	                try:
728	                    hist = get_history(str(symbol), period="5d")
729	                    if hist.empty:
730	                        continue
731	                    _col_map = {c.lower(): c for c in hist.columns}
732	                    close_col = _col_map.get("close", hist.columns[0])
733	                    closes = hist[close_col].values
734	                    if len(closes) < 2:
735	                        continue
736	
737	                    daily_change = float(
738	                        (closes[-1] / closes[-2] - 1) * 100
739	                    )
740	
741	                    if abs(daily_change) > 3:
742	                        alerts.append({
743	                            "symbol": symbol,
744	                            "change_pct": round(daily_change, 2),
745	                            "alert_type": (
746	                                "SURGE" if daily_change > 0 else "DROP"
747	                            ),
748	                            "message": (
749	                                f"{symbol} moved {daily_change:+.1f}% today"
750	                            ),
751	                        })
752	                except Exception:
753	                    continue
754	
755	        except Exception as e:
756	            logger.debug("Watchlist alerts failed: %s", e)
757	
758	        return sorted(
759	            alerts,
760	            key=lambda x: abs(x["change_pct"]),
761	            reverse=True
762	        )[:5]
